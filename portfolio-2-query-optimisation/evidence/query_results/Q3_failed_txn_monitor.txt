-- Q3_failed_txn_monitor
SELECT channel, count(*) AS failed, sum(amount) AS value_ngn
FROM txn
WHERE status = 'FAILED'
  AND created_at >= TIMESTAMPTZ '2026-10-24' AND created_at < TIMESTAMPTZ '2026-11-01'
GROUP BY channel
ORDER BY failed DESC;
