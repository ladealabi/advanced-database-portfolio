-- Q6_last_txn_per_account_correlated
SELECT a.account_number,
       (SELECT max(le.created_at) FROM ledger_entry le WHERE le.account_id = a.account_id) AS last_activity
FROM account a
WHERE a.branch_id = 3 AND a.account_id % 50 = 0;
